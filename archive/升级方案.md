```python
"""
Safe Math Operations
防止除零错误和数值异常
"""

from typing import Optional, Union
from config_extended import SafetyConfig


def safe_divide(numerator: float, denominator: float,
                fallback: Optional[Union[float, str]] = None) -> Union[float, str]:
    """
    安全除法，防止除零错误

    Args:
        numerator: 分子
        denominator: 分母
        fallback: 除零时的返回值（None则使用配置默认值）

    Returns:
        除法结果或fallback值

    Examples:
        >>> safe_divide(100, 50)
        2.0
        >>> safe_divide(100, 0)
        'N/A'
        >>> safe_divide(100, 0, fallback=0)
        0
    """
    if not SafetyConfig.DIVISION_SAFE_MODE:
        return numerator / denominator  # 不安全模式，直接除

    if denominator == 0 or denominator is None:
        return fallback if fallback is not None else SafetyConfig.DIVISION_FALLBACK

    try:
        return numerator / denominator
    except (ZeroDivisionError, TypeError):
        return fallback if fallback is not None else SafetyConfig.DIVISION_FALLBACK


def safe_percentage_change(current: float, previous: float,
                          precision: int = 1) -> Union[float, str]:
    """
    安全计算百分比变化

    Args:
        current: 当前值
        previous: 之前值
        precision: 小数位数

    Returns:
        百分比变化（如 25.5 表示增长25.5%）

    Examples:
        >>> safe_percentage_change(150, 100)
        50.0
        >>> safe_percentage_change(100, 0)
        'N/A'
    """
    result = safe_divide(current - previous, previous, fallback=None)

    if result == SafetyConfig.DIVISION_FALLBACK or result is None:
        return SafetyConfig.DIVISION_FALLBACK

    return round(result * 100, precision)


def format_currency(value: float, currency: str = "$") -> str:
    """
    格式化货币显示

    Args:
        value: 金额
        currency: 货币符号

    Returns:
        格式化后的字符串（如 $1,234.56）
    """
    return f"{currency}{value:,.2f}"


def format_number(value: Union[int, float], precision: int = 0) -> str:
    """
    格式化数字（千位分隔符）

    Args:
        value: 数值
        precision: 小数位数

    Returns:
        格式化后的字符串（如 1,234）
    """
    if precision == 0:
        return f"{int(value):,}"
    else:
        return f"{value:,.{precision}f}"
```

```python
def create_daily_trend_chart(analyzer):
    """创建Daily Trend折线图（趋势线+标注版）"""
    daily_trend = analyzer.get_daily_trend()
    dates = daily_trend.index
    sales = daily_trend.values

    if FeatureFlags.ENABLE_ENHANCED_CHARTS:
        # 主折线
        fig = go.Figure()

        fig.add_trace(go.Scatter(
            x=dates,
            y=sales,
            mode='lines+markers',
            line=dict(
                color=ChartConfig.COLORS['primary'],
                width=ChartConfig.DIMENSIONS['line_width']
            ),
            marker=dict(
                size=ChartConfig.DIMENSIONS['marker_size'],
                color=ChartConfig.COLORS['primary']
            ),
            fill='tozeroy',
            fillcolor=f"rgba(31, 119, 180, 0.2)",
            name='Daily Sales',
            hovertemplate='<b>%{x|%Y-%m-%d}</b><br>销售额: %{y:$,.2f}<extra></extra>'
        ))

        # 添加趋势线（如果scipy可用）
        if SCIPY_AVAILABLE and len(dates) > 2:
            try:
                x_numeric = list(range(len(dates)))
                slope, intercept, r_value, p_value, std_err = stats.linregress(x_numeric, sales)
                trend_line = [slope * x + intercept for x in x_numeric]

                fig.add_trace(go.Scatter(
                    x=dates,
                    y=trend_line,
                    mode='lines',
                    line=dict(dash='dash', color=ChartConfig.COLORS['accent'], width=2),
                    name=f'趋势线 (R²={r_value**2:.2f})',
                    hovertemplate='趋势: %{y:$,.2f}<extra></extra>'
                ))
            except Exception as e:
                print(f"趋势线计算失败: {e}")

        # 标注最高点和最低点
        max_idx = sales.argmax()
        min_idx = sales.argmin()

        # 检查标注是否重叠（距离<20%数据点）
        if abs(max_idx - min_idx) > len(dates) * 0.2:
            # 最高点标注
            fig.add_annotation(
                x=dates[max_idx],
                y=sales[max_idx],
                text=f"最高<br>{format_currency(sales[max_idx])}",
                showarrow=True,
                arrowhead=2,
                ax=0,
                ay=-40,
                bgcolor=ChartConfig.COLORS['success'],
                bordercolor=ChartConfig.COLORS['primary'],
                borderwidth=2
            )

            # 最低点标注
            fig.add_annotation(
                x=dates[min_idx],
                y=sales[min_idx],
                text=f"最低<br>{format_currency(sales[min_idx])}",
                showarrow=True,
                arrowhead=2,
                ax=0,
                ay=40,
                bgcolor=ChartConfig.COLORS['warning'],
                bordercolor=ChartConfig.COLORS['accent'],
                borderwidth=2
            )
        else:
            # 只标注最高点
            fig.add_annotation(
                x=dates[max_idx],
                y=sales[max_idx],
                text=f"峰值 {format_currency(sales[max_idx])}",
                showarrow=True,
                ay=-40
            )

        fig.update_layout(
            title=dict(
                text='Daily Sales Trend',
                font=dict(size=ChartConfig.DIMENSIONS['title_font_size']),
                x=0.5,
                xanchor='center'
            ),
            height=ChartConfig.DIMENSIONS['chart_height'],
            plot_bgcolor=ChartConfig.COLORS['background'],
            paper_bgcolor=ChartConfig.COLORS['background'],
            xaxis=dict(
                title='Date',
                tickformat='%m/%d',
                gridcolor=ChartConfig.COLORS['grid'],
                griddash='dot'
            ),
            yaxis=dict(
                title='Sales ($)',
                tickprefix='$',
                tickformat=',.0f',
                gridcolor=ChartConfig.COLORS['grid'],
                griddash='dot'
            ),
            hovermode='x unified'
        )
```

```python
from typing import List, Dict, Any
from utils.safe_math import safe_divide, safe_percentage_change

def detect_anomalies(self) -> List[Dict[str, Any]]:
    """
    检测销售数据异常并生成分类洞察

    Returns:
        List[Dict]: 洞察列表，每个洞察包含：
            - category: 'opportunity' | 'risk' | 'trend'
            - emoji: 对应emoji
            - title: 洞察标题
            - detail: 详细说明
            - color: 背景色
    """
    insights = []

    if self.df.empty:
        return insights

    # 基础统计
    daily_sales = self.get_daily_trend()
    total_sales = self.get_total_sales()
    total_orders = len(self.df)

    if len(daily_sales) < 1:
        return insights

    mean_sales = daily_sales.mean()
    std_sales = daily_sales.std()

    # ========== 机会类洞察 ==========

    # 洞察1: 销售高峰日
    if std_sales > 0:
        high_threshold = mean_sales + 2 * std_sales
        high_days = daily_sales[daily_sales > high_threshold]

        if not high_days.empty:
            best_day = high_days.idxmax()
            best_amount = high_days.max()
            pct_above = safe_percentage_change(best_amount, mean_sales)

            insights.append({
                'category': 'opportunity',
                'emoji': '🔥',
                'title': f'销售高峰日: {best_day}',
                'detail': f'该日销售额 {format_currency(best_amount)}，比日均高 {pct_above}%。分析该日的营销活动或外部因素，可复制成功经验。',
                'color': ChartConfig.COLORS['success']
            })

    # 洞察2: 高客单价机会
    avg_order_value = safe_divide(total_sales, total_orders)
    if avg_order_value != 'N/A' and avg_order_value > 50:
        insights.append({
            'category': 'opportunity',
            'emoji': '💎',
            'title': f'客单价较高: {format_currency(avg_order_value)}',
            'detail': '平均订单价值超过$50，说明客户购买力强。建议推出高价值产品组合或会员计划。',
            'color': ChartConfig.COLORS['success']
        })

    # 洞察3: 增长趋势
    if len(daily_sales) >= 4:
        first_half_avg = daily_sales[:len(daily_sales)//2].mean()
        second_half_avg = daily_sales[len(daily_sales)//2:].mean()
        growth_rate = safe_percentage_change(second_half_avg, first_half_avg)

        if growth_rate != 'N/A' and growth_rate > 10:
            insights.append({
                'category': 'opportunity',
                'emoji': '📈',
                'title': f'销售持续增长 (+{growth_rate}%)',
                'detail': f'数据后半段销售比前半段增长 {growth_rate}%，趋势向好。建议增加库存和营销投入。',
                'color': ChartConfig.COLORS['success']
            })

    # ========== 风险类洞察 ==========

    # 洞察4: 销售低谷日
    if std_sales > 0:
        low_threshold = mean_sales - 2 * std_sales
        low_days = daily_sales[daily_sales < low_threshold]

        if not low_days.empty:
            worst_day = low_days.idxmin()
            worst_amount = low_days.min()
            pct_below = abs(safe_percentage_change(worst_amount, mean_sales))

            insights.append({
                'category': 'risk',
                'emoji': '⚠️',
                'title': f'销售低谷日: {worst_day}',
                'detail': f'该日销售额仅 {format_currency(worst_amount)}，比日均低 {pct_below}%。需调查原因（库存短缺、竞争对手促销等）。',
                'color': ChartConfig.COLORS['warning']
            })

    # 洞察5: 产品集中度风险
    top_product_sales = self.get_top_products(1).iloc[0]
    all_product_sales = self.df.groupby('Product Name')['Quantity'].sum().sum()
    concentration = safe_percentage_change(top_product_sales, all_product_sales - top_product_sales)

    if concentration != 'N/A' and top_product_sales / all_product_sales > 0.4:
        insights.append({
            'category': 'risk',
            'emoji': '⚠️',
            'title': '产品集中度过高',
            'detail': f'Top 1产品占总销量 {round(top_product_sales/all_product_sales*100)}%。过度依赖单一产品存在风险，建议多元化产品组合。',
            'color': ChartConfig.COLORS['warning']
            })

    # 洞察6: 地区集中度风险
    top_state_sales = self.get_sales_by_state().iloc[0]
    concentration_pct = round(top_state_sales / total_sales * 100)

    if concentration_pct > 50:
        top_state = self.get_sales_by_state().index[0]
        insights.append({
            'category': 'risk',
            'emoji': '🗺️',
            'title': f'地区过度集中: {top_state}',
            'detail': f'{top_state}州占总销售额 {concentration_pct}%。建议拓展其他地区市场，分散区域风险。',
            'color': ChartConfig.COLORS['warning']
        })

    # ========== 趋势类洞察 ==========

    # 洞察7: 周末vs工作日模式
    if 'Date' in self.df.columns:
        self.df['Weekday'] = self.df['Date'].dt.dayofweek
        weekend_sales = self.df[self.df['Weekday'].isin([5, 6])]['Total'].sum()
        weekday_sales = self.df[~self.df['Weekday'].isin([5, 6])]['Total'].sum()

        weekend_days = self.df[self.df['Weekday'].isin([5, 6])]['Date'].nunique()
        weekday_days = self.df[~self.df['Weekday'].isin([5, 6])]['Date'].nunique()

        weekend_avg = safe_divide(weekend_sales, weekend_days)
        weekday_avg = safe_divide(weekday_sales, weekday_days)

        if weekend_avg != 'N/A' and weekday_avg != 'N/A':
            diff_pct = safe_percentage_change(weekend_avg, weekday_avg)

            if diff_pct != 'N/A' and abs(diff_pct) > 20:
                trend_type = "更高" if diff_pct > 0 else "更低"
                insights.append({
                    'category': 'trend',
                    'emoji': '📅',
                    'title': f'周末销售{trend_type} ({abs(diff_pct)}%)',
                    'detail': f'周末日均 {format_currency(weekend_avg)} vs 工作日 {format_currency(weekday_avg)}。可据此优化人员排班和营销活动时间。',
                    'color': ChartConfig.COLORS['info']
                })

    # 洞察8: Top 3产品贡献度
    top3_qty = self.get_top_products(3).sum()
    total_qty = self.df['Quantity'].sum()
    top3_contribution = safe_divide(top3_qty, total_qty) * 100

    if top3_contribution != 'N/A':
        insights.append({
            'category': 'trend',
            'emoji': '🏆',
            'title': f'Top 3产品贡献 {round(top3_contribution)}%',
            'detail': f'前3名产品占总销量 {round(top3_contribution)}%。这些明星产品应重点推广和保证库存。',
            'color': ChartConfig.COLORS['info']
        })

    # 洞察9: 订单量vs销售额趋势
    if len(daily_sales) >= 3:
        daily_orders = self.df.groupby(self.df['Date'].dt.date).size()

        # 计算相关性
        if len(daily_orders) == len(daily_sales):
            try:
                import numpy as np
                correlation = np.corrcoef(daily_sales.values, daily_orders.values)[0, 1]

                if correlation < 0.5:
                    insights.append({
                        'category': 'trend',
                        'emoji': '🔍',
                        'title': '订单量与销售额相关性低',
                        'detail': f'订单数量增加不一定带来销售额增长（相关性 {correlation:.2f}）。可能存在大订单客户，建议分层运营。',
                        'color': ChartConfig.COLORS['info']
                    })
            except:
                pass

    # 洞察10: 价格分布洞察
    avg_price = self.df['Price'].mean()
    median_price = self.df['Price'].median()
    price_std = self.df['Price'].std()

    if price_std / avg_price > 0.5:  # 变异系数 > 0.5
        insights.append({
            'category': 'trend',
            'emoji': '💰',
            'title': '产品价格分布分散',
            'detail': f'价格范围大（均价 {format_currency(avg_price)}，标准差 {format_currency(price_std)}）。建议优化定价策略，考虑价格分层。',
            'color': ChartConfig.COLORS['info']
        })

    # 限制最多返回10条
    return insights[:10]
```

```python
from utils.safe_math import safe_divide, safe_percentage_change, format_currency

def detect_comparison_opportunities(self) -> Dict[str, Any]:
    """
    检测数据是否支持周/月对比

    Returns:
        Dict: {
            'can_compare_weeks': bool,
            'can_compare_months': bool,
            'time_range_days': int,
            'warning': str (如果有警告)
        }
    """
    if self.df.empty or 'Date' not in self.df.columns:
        return {
            'can_compare_weeks': False,
            'can_compare_months': False,
            'time_range_days': 0,
            'warning': '数据为空或缺少日期列'
        }

    date_range_days = (self.df['Date'].max() - self.df['Date'].min()).days

    result = {
        'can_compare_weeks': date_range_days >= 14,
        'can_compare_months': date_range_days >= 60,
        'time_range_days': date_range_days,
        'warning': None
    }

    # 添加警告
    if date_range_days < 14:
        result['warning'] = f'数据仅跨 {date_range_days} 天，不足2周无法对比'
    elif date_range_days < 21:
        result['warning'] = '数据接近2周边界，对比结果仅供参考'

    return result


def get_week_comparison(self) -> Dict[str, Any]:
    """
    本周vs上周对比（最近7天 vs 之前7天）

    Returns:
        Dict: {
            'this_week': {'sales': float, 'orders': int, 'avg_order': float},
            'last_week': {...},
            'changes': {'sales_pct': float, 'orders_pct': float, 'avg_order_pct': float},
            'period_labels': {'this': str, 'last': str}
        }
    """
    if self.df.empty:
        return None

    max_date = self.df['Date'].max()

    # 最近7天
    this_week_start = max_date - pd.Timedelta(days=6)
    this_week_data = self.df[self.df['Date'] >= this_week_start]

    # 之前7天
    last_week_start = this_week_start - pd.Timedelta(days=7)
    last_week_end = this_week_start - pd.Timedelta(days=1)
    last_week_data = self.df[(self.df['Date'] >= last_week_start) & (self.df['Date'] <= last_week_end)]

    # 如果上周数据不足7天，返回None
    if len(last_week_data) == 0:
        return None

    # 计算指标
    this_week_sales = this_week_data['Total'].sum()
    this_week_orders = len(this_week_data)
    this_week_avg = safe_divide(this_week_sales, this_week_orders)

    last_week_sales = last_week_data['Total'].sum()
    last_week_orders = len(last_week_data)
    last_week_avg = safe_divide(last_week_sales, last_week_orders)

    # 计算变化（使用安全函数）
    sales_change = safe_percentage_change(this_week_sales, last_week_sales)
    orders_change = safe_percentage_change(this_week_orders, last_week_orders)

    if this_week_avg != 'N/A' and last_week_avg != 'N/A':
        avg_change = safe_percentage_change(this_week_avg, last_week_avg)
    else:
        avg_change = 'N/A'

    return {
        'this_week': {
            'sales': this_week_sales,
            'orders': this_week_orders,
            'avg_order': this_week_avg
        },
        'last_week': {
            'sales': last_week_sales,
            'orders': last_week_orders,
            'avg_order': last_week_avg
        },
        'changes': {
            'sales_pct': sales_change,
            'orders_pct': orders_change,
            'avg_order_pct': avg_change
        },
        'period_labels': {
            'this': f'{this_week_start.strftime("%m/%d")} - {max_date.strftime("%m/%d")}',
            'last': f'{last_week_start.strftime("%m/%d")} - {last_week_end.strftime("%m/%d")}'
        }
    }


def get_month_comparison(self) -> Dict[str, Any]:
    """
    本月vs上月对比（最近30天 vs 之前30天）

    Returns:
        Dict: 同get_week_comparison结构
    """
    if self.df.empty:
        return None

    max_date = self.df['Date'].max()

    # 最近30天
    this_month_start = max_date - pd.Timedelta(days=29)
    this_month_data = self.df[self.df['Date'] >= this_month_start]

    # 之前30天
    last_month_start = this_month_start - pd.Timedelta(days=30)
    last_month_end = this_month_start - pd.Timedelta(days=1)
    last_month_data = self.df[(self.df['Date'] >= last_month_start) & (self.df['Date'] <= last_month_end)]

    if len(last_month_data) == 0:
        return None

    # 计算指标（同周对比逻辑）
    this_month_sales = this_month_data['Total'].sum()
    this_month_orders = len(this_month_data)
    this_month_avg = safe_divide(this_month_sales, this_month_orders)

    last_month_sales = last_month_data['Total'].sum()
    last_month_orders = len(last_month_data)
    last_month_avg = safe_divide(last_month_sales, last_month_orders)

    sales_change = safe_percentage_change(this_month_sales, last_month_sales)
    orders_change = safe_percentage_change(this_month_orders, last_month_orders)
    avg_change = safe_percentage_change(this_month_avg, last_month_avg) if this_month_avg != 'N/A' and last_month_avg != 'N/A' else 'N/A'

    return {
        'this_month': {
            'sales': this_month_sales,
            'orders': this_month_orders,
            'avg_order': this_month_avg
        },
        'last_month': {
            'sales': last_month_sales,
            'orders': last_month_orders,
            'avg_order': last_month_avg
        },
        'changes': {
            'sales_pct': sales_change,
            'orders_pct': orders_change,
            'avg_order_pct': avg_change
        },
        'period_labels': {
            'this': f'{this_month_start.strftime("%m/%d")} - {max_date.strftime("%m/%d")}',
            'last': f'{last_month_start.strftime("%m/%d")} - {last_month_end.strftime("%m/%d")}'
        }
```

```python
def show_comparison_section(analyzer):
    """显示周/月对比section"""
    if not FeatureFlags.ENABLE_COMPARISON_FEATURE:
        return

    # 检测对比机会
    opportunities = analyzer.detect_comparison_opportunities()

    if not opportunities['can_compare_weeks'] and not opportunities['can_compare_months']:
        if opportunities['warning']:
            st.info(f"ℹ️ {opportunities['warning']}")
        return

    st.markdown("---")
    st.subheader("📊 时间对比分析")

    # 让用户选择对比类型
    compare_options = []
    if opportunities['can_compare_weeks']:
        compare_options.append('week')
    if opportunities['can_compare_months']:
        compare_options.append('month')

    if len(compare_options) == 0:
        return

    # 默认选择最合适的
    default_option = 'month' if 'month' in compare_options else 'week'

    compare_type = st.radio(
        "选择对比周期",
        options=compare_options,
        format_func=lambda x: "📅 周对比（最近7天 vs 之前7天）" if x == 'week' else "📆 月对比（最近30天 vs 之前30天）",
        index=compare_options.index(default_option),
        horizontal=True
    )

    # 获取对比数据
    if compare_type == 'week':
        comparison = analyzer.get_week_comparison()
        period_name = "周"
    else:
        comparison = analyzer.get_month_comparison()
        period_name = "月"

    if comparison is None:
        st.warning(f"⚠️ 数据不足，无法进行{period_name}对比")
        return

    if opportunities['warning']:
        st.warning(f"⚠️ {opportunities['warning']}")

    # 显示对比卡片
    col1, col2, col3 = st.columns(3)

    metrics = [
        {
            'label': '销售额',
            'current': comparison[f'this_{compare_type}']['sales'],
            'previous': comparison[f'last_{compare_type}']['sales'],
            'change': comparison['changes']['sales_pct'],
            'format': 'currency'
        },
        {
            'label': '订单量',
            'current': comparison[f'this_{compare_type}']['orders'],
            'previous': comparison[f'last_{compare_type}']['orders'],
            'change': comparison['changes']['orders_pct'],

        },

            'change': comparison['changes']['avg_order_pct'],
            'format': 'currency'
        }
    ]

    for col, metric in zip([col1, col2, col3], metrics):
        with col:
            display_comparison_card(
                metric['label'],
                metric['current'],
                metric['previous'],
                metric['change'],
                comparison['period_labels'],
                metric['format']
            )


def display_comparison_card(label, current, previous, change_pct, period_labels, format_type='currency'):
    """显示单个对比卡片"""
    # 格式化数值
    if format_type == 'currency':
        if current != 'N/A':
            current_str = format_currency(current)
        else:
            current_str = 'N/A'
        if previous != 'N/A':
            previous_str = format_currency(previous)
        else:
            previous_str = 'N/A'
    else:
        current_str = format_number(current) if current != 'N/A' else 'N/A'
        previous_str = format_number(previous) if previous != 'N/A' else 'N/A'

    # 变化方向
    if change_pct == 'N/A' or change_pct is None:
        arrow = ""
        color = "#666"
        change_str = "N/A"
    elif change_pct > 0:
        arrow = "↑"
        color = "#28a745"  # 绿色
        change_str = f"+{change_pct}%"
    elif change_pct < 0:
        arrow = "↓"
        color = "#dc3545"  # 红色
        change_str = f"{change_pct}%"
    else:
        arrow = "→"
        color = "#666"
        change_str = "0%"

    st.markdown(f"""
        <div class="saas-card" style="text-align: center; padding: 20px;">
            <div style="color: #86868B; font-size: 0.9rem; margin-bottom: 10px;">
                {label}
            </div>
            <div style="font-size: 2rem; font-weight: 700; color: {ChartConfig.COLORS['primary']}; margin: 10px 0;">
                {current_str}
            </div>
            <div style="font-size: 1.5rem; font-weight: 600; color: {color}; margin: 10px 0;">
                {arrow} {change_str}
            </div>
            <div style="color: #86868B; font-size: 0.8rem; margin-top: 10px;">
                上期: {previous_str}<br>
                <span style="font-size: 0.7rem;">本期 {period_labels['this']} vs 上期 {period_labels['last']}</span>
            </div>
        </div>
    """, unsafe_allow_html=True)
```
